export function useGatewayBoot(opts: BootOptions) { const backoff = createBackoff({ base: 250, max: 8000 }) const socket = openGatewaySocket(opts.url) // Revoked tokens: fail fast and raise the sign-in prompt. socket.on('event', event => { if (event.code === 'auth_required') { $needsLogin.set(true) return stopReconnect('revoked') } }) socket.on('close', () => { if (!$needsLogin.get()) scheduleReconnect(backoff.next()) }) onCleanup(() => stopReconnect('unmount')) return { socket, backoff } }
